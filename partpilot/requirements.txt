# --- Brain 2: similarity search ---------------------------------------------------------------
# Embedding backends are swappable (see EMBEDDING_BACKEND): open_clip_torch
# provides OpenCLIP, and transformers below provides DINOv2 and SigLIP.
# torch itself is imported directly (embedding_backends.py, clip_model.py),
# not just pulled in transitively by the two packages above, so it is pinned
# explicitly rather than left to whatever version they happen to resolve.
torch==2.13.0
open_clip_torch==2.29.0
faiss-cpu==1.9.0.post1

# --- Brain 4: reasoning (Qwen) + Brain 2 backends (DINOv2, SigLIP) ---------------------------------------------------------------
transformers==4.46.3
accelerate==1.1.1
# LLM_BACKEND=llamacpp (the default) needs this. A plain `pip install
# llama-cpp-python` tries to build from source and fails outright on Windows
# without Long Path support enabled (a common default) - its sdist contains
# a llama.cpp vendor path deep enough to exceed MAX_PATH. The
# --extra-index-url at the top of this file avoids that entirely by
# fetching a prebuilt wheel instead (verified: installs clean even without
# Long Path support). Brain 4 is still optional at runtime either way - a
# failed load just degrades to the Brain 1-3 answer with no explanation -
# so set LLM_BACKEND=transformers instead if this ever fails to install.
llama-cpp-python==0.3.34
